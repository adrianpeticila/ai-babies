# ===================================================================== # BP-02: PRODUCTION HYBRID RAG WITH RRF & CROSS-ENCODER (v1.2.0) # ===================================================================== from typing import List, Dict, Any import numpy as np class HybridRRFRetriever: def __init__(self, bm25_index, vector_index, reranker_client, k: int = 60): self.bm25 = bm25_index self.vector = vector_index self.reranker = reranker_client self.k = k async def retrieve(self, query: str, top_k: int = 5, fetch_k: int = 30) -> List[Dict[str, Any]]: # 1. Parallel fetch from Sparse BM25 and Dense Vector indices sparse_hits = await self.bm25.query(query, limit=fetch_k) dense_hits = await self.vector.query(query, limit=fetch_k) # 2. Reciprocal Rank Fusion calculation scores: Dict[str, float] = {} doc_map: Dict[str, Dict[str, Any]] = {} for rank, hit in enumerate(sparse_hits, start=1): doc_id = hit["id"] scores[doc_id] = scores.get(doc_id, 0.0) + 1.0 / (self.k + rank) doc_map[doc_id] = hit for rank, hit in enumerate(dense_hits, start=1): doc_id = hit["id"] scores[doc_id] = scores.get(doc_id, 0.0) + 1.0 / (self.k + rank) doc_map[doc_id] = hit # 3. Sort by fused score and take top candidates for Cross-Encoder sorted_doc_ids = sorted(scores.keys(), key=lambda x: scores[x], reverse=True)[:fetch_k] candidates = [doc_map[did] for did in sorted_doc_ids] # 4. Cross-Encoder Re-Ranking reranked_results = await self.reranker.rank(query=query, documents=candidates, top_n=top_k) return reranked_results
